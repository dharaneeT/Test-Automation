package Pages;

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.support.ui.Select;

public class EmployeeListPage {

	private WebDriver driver;

	public EmployeeListPage(WebDriver driver) {
		this.driver = driver;
	}


	private By newEmpLnk = By.linkText("+ New Employee");
	private By nameField = By.name("searchTerm");
	private By emailField = By.name("emailTerm");
	private By searchBtn = By.xpath("//button[text()='Search']");
	private By clearLnk = By.linkText("✕ Clear");
	private By gradeFilter = By.name("gradeFilter");

	public CreateEmpPage createEmp() {
		driver.findElement(newEmpLnk).click();
		return new CreateEmpPage(driver);
	}

	public void findEmp(String name, String mail, int gradeindex) throws InterruptedException {
		driver.findElement(nameField).sendKeys(name);
		driver.findElement(emailField).sendKeys(mail);

		Select gradeSelect = new Select(driver.findElement(gradeFilter));
		gradeSelect.selectByIndex(gradeindex);

		driver.findElement(searchBtn).click();

		Thread.sleep(5000);
	}

	public void clearList() {
		driver.findElement(clearLnk).click();
	}
}
